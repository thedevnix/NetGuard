cmake_minimum_required(VERSION 3.16)
project(NetGuard VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(netguard src/main.cpp)
target_compile_options(netguard PRIVATE -Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
target_link_libraries(netguard PRIVATE Threads::Threads)
enable_testing()
add_test(NAME smoke COMMAND netguard --help)
